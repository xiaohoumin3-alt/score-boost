## 动态 θ 估计（替代固定初始值）

```javascript
// 旧方案（删除）
const gradeAbility = { '1': -1.5, '2': -1.2, ... };

// 新方案：基于前几题正确率动态估计
function getInitialTheta(responses) {
  if (responses.length < 5) {
    // 前 5 题：用正确率推算
    const correctRate = responses.filter(r => r.correct).length / Math.max(1, responses.length);
    return rateToTheta(correctRate);
  } else {
    // 5 题后：用 IRT 估计
    return estimateAbility(responses).theta;
  }
}
```
